## 辅助函数
```bash
wait_apt_lock() {
    echo "ℹ️ 等待apt锁释放，检测/var/lib/apt/lists/lock"
    local max_wait=30
    local cnt=0
    LOCKFILE="/var/lib/apt/lists/lock"

    while [ -f "${LOCKFILE}" ]; do
        # apt lock内部存储持有锁的pid
        LOCK_PID=$(cat "${LOCKFILE}" 2>/dev/null || true)
        # 判断pid是否还在运行
        if [ -n "${LOCK_PID}" ] && kill -0 "${LOCK_PID}" 2>/dev/null; then
            # pid存活：真的有apt进程在跑，等待
            echo "ℹ️ apt进程 ${LOCK_PID} 正在运行，等待释放锁"
        else
            # pid不存在：孤儿残留锁文件，可以安全删除
            echo "⚠️ 发现孤儿apt锁文件，进程已消亡，清理锁"
            sudo rm -f "${LOCKFILE}"
            break
        fi

        sleep 1
        cnt=$((cnt+1))
        if [ $cnt -ge $max_wait ]; then
            echo "⚠️ apt锁等待超时，继续尝试执行apt操作"
            break
        fi
    done
    echo "✅ apt锁处理完成，可以执行apt操作"
}

check_npm_global_cmd() {
    local cmd_name="$1"
    local bin_path="${NPM_GLOBAL_BIN}/${cmd_name}"
    if [ ! -x "${bin_path}" ]; then
        echo "❌ npm全局命令 ${cmd_name} 缺失，预期路径: ${bin_path}" >&2
        exit 1
    fi
    echo "✅ ${cmd_name} 存在: ${bin_path}"
    echo ""
}
```

## 安装 Node.js LTS22
```bash
echo "===== 安装 Node.js LTS22 ====="
if command -v node &>/dev/null; then
    echo "=== 检测到 node 已安装，跳过安装流程 ==="
else
    curl -fsSL https://deb.nodesource.com/setup_22.x | sudo bash -
    sudo apt install -y nodejs
fi

node -v
npm -v
echo "Node 版本: $(node -v)"
echo "NPM 版本: $(npm -v)"

NPM_USER_GLOBAL="${HOME}/.npm-global"
mkdir -p "${NPM_USER_GLOBAL}"
npm config set prefix "${NPM_USER_GLOBAL}"
NPM_GLOBAL_BIN="${NPM_USER_GLOBAL}/bin"
FINAL_PATH_LINE=$(printf 'export PATH=%s:$PATH' "${NPM_GLOBAL_BIN}")

# 处理.bashrc：不存在则新建
if [ ! -f "${HOME}/.bashrc" ]; then
    printf "%s\n" "${FINAL_PATH_LINE}" > "${HOME}/.bashrc"
else
    if ! grep -Fxq "${FINAL_PATH_LINE}" "${HOME}/.bashrc"; then
        printf "%s\n" "${FINAL_PATH_LINE}" >> "${HOME}/.bashrc"
    fi
fi

export PATH="${NPM_GLOBAL_BIN}:$PATH"
echo "npm全局安装目录配置完成：$(npm config get prefix)"
```


## 安装git&git cli
```bash
if command -v git &>/dev/null && command -v gh &>/dev/null; then
    echo "=== 检测到 git、gh‑cli 均已安装，跳过安装流程 ==="
    git --version
    gh --version
    echo "✅ git + gh‑cli 已是就绪状态"
else
    echo "=== apt 更新，安装 git、ca‑certificates、curl、gnupg ==="
    sudo apt install -y git ca-certificates curl gnupg
    echo "=== 配置 GitHub CLI apt源 ==="
    curl -fsSL https://cli.github.com/packages/githubcli-archive-keyring.gpg | sudo gpg --batch --yes --dearmor -o /usr/share/keyrings/githubcli-archive-keyring.gpg
    sudo chmod go+r /usr/share/keyrings/githubcli-archive-keyring.gpg
    echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/githubcli-archive-keyring.gpg] https://cli.github.com/packages stable main" | sudo tee /etc/apt/sources.list.d/github-cli.list > /dev/null
    sudo apt install -y gh
    git --version
    gh --version
    echo "✅ git + gh‑cli 安装完成"
fi

test_github_ssh_podman() {
    echo "==== podman DEBUG ssh 测试开始 ====" >&2
    local out
    out=$(ssh -T \
        -o StrictHostKeyChecking=no \
        -o UserKnownHostsFile="${HOME}/.ssh/known_hosts" \
        -o ConnectTimeout=10 \
        git@github.com 2>&1)
    local ret=$?
    echo "$out" >&2
    if echo "$out" | grep -q "successfully authenticated"; then
        return 0
    else
        return 1
    fi
}

echo "容器内 SSH_AUTH_SOCK = ${SSH_AUTH_SOCK:-未设置}"
MAX_RETRY=3
RETRY_DELAY=2
ok=1
for ((i=0;i<MAX_RETRY;i++)); do
    if test_github_ssh_podman ; then
        ok=0
        break
    fi
    echo "⚠️ GitHub SSH测试失败，等待${RETRY_DELAY}秒后重试 ($((i+1))/${MAX_RETRY})" >&2
    sleep ${RETRY_DELAY}
done
if [[ ${ok} -ne 0 ]]; then
    echo "ERROR: 容器内经过${MAX_RETRY}次重试，仍然无法连通GitHub SSH，请检查socket挂载、宿主机ssh‑agent是否加载私钥、网络连接" >&2
    exit 1
fi
echo "容器内：GitHub SSH测试通过"
```
